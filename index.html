async function parseLabelWithGemini(base64Image, apiKey) { const prompt = `Jesteś precyzyjnym skanerem OCR dla logistyki magazynowej. Przeanalizuj etykietę i wyodrębnij: - model (dokładny model/nazwa sprzętu) - serial (numer seryjny, S/N) - mac (adres fizyczny MAC, zamień na format z dwukropkami XX:XX:XX:XX:XX:XX jeśli brak separatorów) - extra (inne widoczne parametry: Part Number, P/N, rewizja) Zwróć TYLKO poprawny format JSON wg schematu: {"model": "", "serial": "", "mac": "", "extra": ""}`; // Użycie modelu gemini-2.5-flash const endpoint = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${apiKey}`; try { const response = await fetch(endpoint, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [ { text: prompt }, { inline_data: { mime_type: "image/jpeg", data: base64Image } } ] }], generationConfig: { response_mime_type: "application/json" } }) }); const result = await response.json(); if (result.error) throw new Error(result.error.message); // Oczyszczenie tekstu na wypadek znaczników Markdown ```json ... ``` let rawText = result.candidates[0].content.parts[0].text; rawText = rawText.replace(/```json/gi, '').replace(/```/g, '').trim(); const parsed = JSON.parse(rawText); document.getElementById('modelField').value = parsed.model || ''; document.getElementById('serialField').value = parsed.serial || ''; document.getElementById('macField').value = parsed.mac || ''; document.getElementById('extraField').value = parsed.extra || ''; saveBtn.disabled = false; showStatus('Dane odczytane! Sprawdź poprawność i kliknij Zapisz.', 'status-success'); } catch (err) { showStatus('Błąd odczytu AI: ' + err.message, 'status-error'); saveBtn.disabled = false; } }
